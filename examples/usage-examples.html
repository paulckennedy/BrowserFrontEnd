🚀 Markdown Editor Component Usage Examples

This page demonstrates how to use the reusable Markdown Editor component in different scenarios and frameworks.

1. Basic Usage - Vanilla HTML

Simply include the component script and use the custom element:

3. React Integration

How to use the component in a React application:

import React, { useRef, useEffect, useState } from 'react'; import 'path/to/markdown-editor-component.js'; function MarkdownEditorReact({ initialContent, onContentChange }) { const editorRef = useRef(null); const [content, setContent] = useState(initialContent || ''); useEffect(() => { const editor = editorRef.current; if (editor) { // Set initial content editor.setContent(content); // Listen for content changes const handleContentChange = (e) => { const newContent = e.detail.content; setContent(newContent); onContentChange?.(newContent); }; editor.addEventListener('content-change', handleContentChange); return () => { editor.removeEventListener('content-change', handleContentChange); }; } }, []); return ( <markdown-editor ref={editorRef} mode="wysiwyg" height="400px" autosave="true" /> ); } export default MarkdownEditorReact;

4. Vue.js Integration

Using the component in Vue.js:

<template> <div> <markdown-editor ref="editor" :mode="mode" :theme="theme" :height="height" @content-change="handleContentChange" @save="handleSave" /> <button @click="saveContent">Save</button> <button @click="loadContent">Load</button> </div> </template> <script> import 'path/to/markdown-editor-component.js'; export default { name: 'MarkdownEditorVue', data() { return { mode: 'wysiwyg', theme: 'default', height: '400px', content: '' }; }, methods: { handleContentChange(event) { this.content = event.detail.content; this.$emit('content-change', this.content); }, handleSave(event) { console.log('Saving content:', event.detail.content); // Implement your save logic }, saveContent() { const content = this.$refs.editor.save(); console.log('Manual save:', content); }, loadContent() { const sampleContent = '# Loaded Content\n\nThis content was loaded dynamically.'; this.$refs.editor.setContent(sampleContent); } } }; </script>

5. JavaScript API Usage

Programmatic control of the editor:

// Get reference to the editor const editor = document.getElementById('api-editor'); // Wait for editor to be ready editor.addEventListener('editor-ready', () => { console.log('Editor is ready!'); }); // Listen for content changes editor.addEventListener('content-change', (e) => { console.log('Content changed:', e.detail.content); }); // Programmatic control editor.setContent('# API Control\n\nThis content was set via JavaScript API.'); editor.configure({ theme: 'dark', mode: 'markdown' }); editor.insertText('\n\nAdditional text inserted!'); editor.focus(); // Get current content const content = editor.getContent();

6. Framework Integration Guide

Angular Integration:

// app.module.ts import { CUSTOM_ELEMENTS_SCHEMA } from '@angular/core'; @NgModule({ schemas: [CUSTOM_ELEMENTS_SCHEMA], // ... other config }) // component.ts import { Component, ViewChild, ElementRef } from '@angular/core'; import 'path/to/markdown-editor-component.js'; @Component({ selector: 'app-editor', template: ` <markdown-editor #editor mode="wysiwyg" (content-change)="onContentChange($event)"> </markdown-editor> ` }) export class EditorComponent { @ViewChild('editor') editor!: ElementRef; onContentChange(event: any) { console.log('Content:', event.detail.content); } }

Svelte Integration:

<script> import 'path/to/markdown-editor-component.js'; let editorElement; let content = ''; function handleContentChange(event) { content = event.detail.content; } </script> <markdown-editor bind:this={editorElement} mode="wysiwyg" height="400px" on:content-change={handleContentChange} />
